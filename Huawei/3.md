# 3、最赚钱的骑手

外卖骑手是一个和时间赛跑的工作。由于外卖订单一般集中在一日三餐的时间,具有时间短突发量大的特点,
骑手们需要在订单集中的时间段内尽可能送出更多的订单以获得更多的收入。
骑手的工作流程大致是这样的,从一个商家出发,选择一个订单,送到目的地点,在目的地点接新的订单送到
下一个目的地,如此循环。一个订单只能配送一次,骑手不能重复配送相同的订单。
从出发地点到目的地点的一次配送需要消耗1单位的时间。同一个地点可能发出多笔订单,其中一些订单可能要
送往同一个目的地,这些起点和终点相同的订单可以合并配送,合并配送的多笔订单和单笔订单一样都只消耗1
单位的时间。
简化起见,我们不区分商家和配送目的地点,统一视作地图上的一个地点。骑手的工作时间总时间是给定的,
如果剩余的工作时间为0,则配送工作结束;如果到达目的地点之后,没有再从该地点发出的订单,则配送工作
结束。

假设地图上有N个地点,每个地点有[0, X]个待送出的订单,每个订单以[s,d,f]来表示,一共有M个订
单。其中s是出发站点编号,d是目的站点编号,p是这笔订单的配送费,约定s!= d。给定骑手的出发
地点编号 $S_n$ 和工作总时间t,请计算出骑手最多能获得的收入。
具体每个取值的范围说明:

| 取值  | 范围      |
| ----- | --------- |
| M     | (0, 104 ] |
| s     | [0, N)    |
| d     | [0, N)    |
| p     | (0, 104 ] |
| $S_n$ | [0, N)    |
| t     | [1, 104)  |

```cpp
#include <bits/stdc++.h>

// 订单结构体，表示一个配送订单
struct Order {
    size_t src;    // 订单起点位置
    size_t dst;    // 订单终点位置
    size_t profit; // 订单配送费用

    Order(size_t src, size_t dst, size_t profit) : src(src), dst(dst), profit(profit) {}
};

/**
 * 订单比较函数，用于排序
 * 按照起点、终点、利润的顺序进行比较
 * 这个排序是为了后续合并相同起点和终点的订单
 */
bool cmp(const Order& a, const Order& b) {
    if (a.src != b.src)
        return a.src < b.src;
    if (a.dst != b.dst)
        return a.dst < b.dst;
    return a.profit < b.profit;
}

/**
 * 合并订单函数
 * 将起点和终点相同的订单合并为一个订单，利润为所有订单利润之和
 * 这符合题目中"起点和终点相同的订单可以合并配送"的要求
 */
std::vector<Order> CombineOrders(const std::vector<Order>& orders) {
    std::vector<Order> ordered(orders); // 复制原始订单列表
    std::vector<Order> res;             // 存储合并后的结果
    
    // 按照起点和终点排序
    std::sort(ordered.begin(), ordered.end(), cmp);
    size_t i = 0;
    while (i < ordered.size()) {
        size_t j = i + 1;
        size_t totalProfit = ordered[i].profit; // 初始利润为第一个订单的利润
        
        // 寻找所有起点和终点都相同的订单，累加它们的利润
        while (j < ordered.size() && ordered[j].src == ordered[i].src && ordered[j].dst == ordered[i].dst) {
            totalProfit += ordered[j].profit;
            j++;
        }
        
        // 创建一个新的合并订单，利润为所有相同起终点订单的总利润
        res.emplace_back(ordered[i].src, ordered[i].dst, totalProfit);
        i = j; // 移动到下一组订单
    }
    return res;
}

/**
 * 计算骑手能获得的最大利润及其路径
 * 使用动态规划算法求解
 * @param start_location 骑手的起始位置
 * @param time 骑手的总工作时间
 * @param orders 所有可用的订单列表
 * @return 包含最大利润和对应路径的pair
 */
std::pair<size_t, std::vector<size_t>> calc_total_profit(size_t start_location, int64_t time, const std::vector<Order>& orders) {
    // 首先合并所有起点和终点相同的订单
    std::vector<Order> combinedOrders = CombineOrders(orders);
    
    // 找出所有地点的数量（通过查找最大的地点编号）
    size_t numLocations = 0;
    for (const auto& order : combinedOrders) {
        numLocations = std::max(numLocations, std::max(order.src, order.dst) + 1);
    }
    
    // 建立订单图：每个位置可以到达的其他位置及对应的利润
    // orderGraph[i] 表示从位置i出发可以送达的所有订单
    std::vector<std::vector<std::pair<size_t, size_t>>> orderGraph(numLocations);
    for (const auto& order : combinedOrders) {
        orderGraph[order.src].push_back({order.dst, order.profit});
    }
    
    // 动态规划状态定义：
    // dp[loc][t] 表示在时间t到达位置loc所能获得的最大收入
    std::vector<std::vector<size_t>> dp(numLocations, std::vector<size_t>(time + 1, 0));
    
    // 记录最优决策的来源，用于后续重建路径
    // parent[loc][t] = {prev_loc, prev_time} 表示到达位置loc的时间t的最优状态是从prev_loc在prev_time时刻转移而来
    std::vector<std::vector<std::pair<size_t, size_t>>> parent(numLocations, std::vector<std::pair<size_t, size_t>>(time + 1, {-1, -1}));
    
    // 初始状态：骑手在起始位置，时间为0，收入为0
    dp[start_location][0] = 0;
    
    // 动态规划求解 - 状态转移
    for (size_t t = 0; t < time; ++t) {
        for (size_t loc = 0; loc < numLocations; ++loc) {
            // 只有当前位置在当前时间有效（有收入或是起始位置）才考虑从这里出发
            if (dp[loc][t] > 0 || loc == start_location) {
                // 遍历从当前位置可以配送的所有订单
                for (const auto& [dst, profit] : orderGraph[loc]) {
                    // 状态转移方程：如果从loc到dst能获得更高的收入，则更新状态
                    // t+1表示配送需要1单位时间
                    if (t + 1 <= time && dp[dst][t + 1] < dp[loc][t] + profit) {
                        dp[dst][t + 1] = dp[loc][t] + profit; // 更新最大收入
                        parent[dst][t + 1] = {loc, t};        // 记录来源状态
                    }
                }
            }
        }
    }
    
    // 找出最大收入和对应的终点和时间
    // 由于骑手可以在任意时间和位置结束工作，需要遍历所有可能的结束状态
    size_t maxProfit = 0;
    size_t endLoc = 0;   // 最优路径的终点
    size_t endTime = 0;  // 最优路径的结束时间
    
    for (size_t loc = 0; loc < numLocations; ++loc) {
        for (size_t t = 0; t <= time; ++t) {
            if (dp[loc][t] > maxProfit) {
                maxProfit = dp[loc][t];
                endLoc = loc;
                endTime = t;
            }
        }
    }
    
    // 回溯构建最优路径
    std::vector<size_t> path;
    if (maxProfit > 0) { // 只有在有收入的情况下才构建路径
        size_t currLoc = endLoc;   // 从终点开始回溯
        size_t currTime = endTime;
        
        // 从终点回溯到起点
        while (currTime > 0) {
            path.push_back(currLoc); // 将当前位置加入路径
            auto [prevLoc, prevTime] = parent[currLoc][currTime]; // 获取前一个状态
            currLoc = prevLoc;   // 更新当前位置
            currTime = prevTime; // 更新当前时间
        }
        
        path.push_back(start_location); // 添加起始位置
        std::reverse(path.begin(), path.end()); // 反转路径，从起点到终点
    }
    
    return {maxProfit, path}; // 返回最大利润和对应的路径
}

/**
 * 主函数
 * 处理输入数据，调用算法函数，输出结果
 */
int main() {
    // 读取骑手的起始位置和总工作时间
    size_t start_location; // 骑手的起始位置
    int64_t time;          // 骑手的总工作时间
    std::cin >> start_location >> time;

    // 读取订单数量
    size_t m;  // 订单总数
    std::cin >> m;

    // 读取所有订单信息
    std::vector<Order> orders;
    for (size_t i = 0; i < m; ++i) {
        size_t src, dst, profit;  // 订单的起点、终点和利润
        std::cin >> src >> dst >> profit;
        orders.emplace_back(src, dst, profit);  // 添加到订单列表
    }   
    
    // 调用动态规划算法计算最大利润和路径
    auto [total_profit, path] = calc_total_profit(start_location, time, orders);
    
    // 输出最大利润
    std::cout << total_profit << std::endl;
    
    // 输出最优路径
    // 路径表示骑手依次经过的位置
    for (size_t i = 0; i < path.size(); ++i) {
        std::cout << path[i];
        if (i != path.size() - 1) std::cout << " ";  // 位置之间用空格分隔
    }

    return 0;
}
```
